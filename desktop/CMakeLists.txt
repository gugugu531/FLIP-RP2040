cmake_minimum_required(VERSION 3.16)
project(flip_rp2040_desktop C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
add_library(fixed_math ../src/fixed_math.c)
add_library(fluid ../src/fluid.c)
add_library(renderer ../src/render.c)
target_link_libraries(fluid PUBLIC fixed_math)
target_link_libraries(renderer PUBLIC fixed_math)
foreach(target fixed_math fluid renderer)
    target_include_directories(${target} PUBLIC ../include)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    if(SANITIZE)
        target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PUBLIC -fsanitize=address,undefined)
    endif()
endforeach()
find_package(SDL2 QUIET)
if(SDL2_FOUND)
    add_executable(flip_rp2040_preview preview.c)
    target_include_directories(flip_rp2040_preview PRIVATE ${SDL2_INCLUDE_DIRS})
    target_link_libraries(flip_rp2040_preview fluid renderer ${SDL2_LIBRARIES} m)
endif()
add_executable(flip_rp2040_capture capture.c)
target_link_libraries(flip_rp2040_capture fluid renderer m)
option(BUILD_TESTING "Build the optional solver/render regression checks" OFF)
include(CTest)
if(BUILD_TESTING)
    add_subdirectory(../tests tests)
endif()
